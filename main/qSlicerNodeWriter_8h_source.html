73 bool write(const qSlicerIO::IOProperties& properties) override
74 {
75 vtkMRMLNodeWriter* writer = this->nodeWriter();
76 if (!writer)
77 {
78 return false;
79 }
80 writer->ClearWrittenNodeIDs();
81 writer->SetScene(this->mrmlScene());
82 vtkMRMLStorableNode* node = vtkMRMLStorableNode::SafeDownCast(this->getNodeByID(properties.value("nodeID").toString().toUtf8().constData()));
83 if (!node || this->canWriteObjectConfidence(node) <= 0.0)
84 {
85 return false;
86 }
87 vtkNew<vtkMRMLIOProperties> vtkProperties;
88 qSlicerIO::toVTKProperties(properties, vtkProperties);
89 if (node->GetID() && writer->GetNodeByID(node->GetID()) == node && writer->CanWriteObjectConfidence(node) > 0.0)
90 {
91 // The VTK-based writer can write the node: use Write(), which may be overridden in vtkMRMLNodeWriter subclasses
92 return writer->Write(vtkProperties);
93 }
94 // This writer accepts a node that the VTK-based writer would not (getNodeByID() or canWriteObject() is overridden)
95 return writer->WriteNode(node, vtkProperties);
96 }
