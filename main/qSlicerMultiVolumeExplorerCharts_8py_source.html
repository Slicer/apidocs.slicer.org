342 self.baselineAverageSignal = 0
343 image = multiVolumeNode.GetImageData()

347 val = image.GetScalarComponentAsDouble(ijk[0], ijk[1], ijk[2], c)

462 def computePercentageChangeWithRespectToBaseline(self):
463 nComponents = self.multiVolumeNode.GetNumberOfFrames()
464 nBaselines = min(self.baselineFrames.value, nComponents)
465 for k in self.labeledVoxels.keys():
466 arr = self.dataNodes[k].GetArray()
467 baseline = 0
468 for bc in range(nBaselines):
469 baseline += arr.GetComponent(bc, 1)
470 baseline /= nBaselines
471 if baseline != 0:
472 for ic in range(nComponents):
473 intensity = arr.GetComponent(ic, 1)
474 percentChange = int(intensity / baseline - 1) * 100.
475 arr.SetComponent(ic, 1, percentChange)
