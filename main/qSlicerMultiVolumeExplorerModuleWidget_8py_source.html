214 def refreshObservers(self):
215 """When the layout changes, drop the observers from
216 all the old widgets and create new observers for the
217 newly created widgets
218 """
219 self.removeObservers()
220 # get new slice nodes
221 layoutManager = slicer.app.layoutManager()
222 sliceNodeCount = slicer.mrmlScene.GetNumberOfNodesByClass("vtkMRMLSliceNode")
223 for nodeIndex in range(sliceNodeCount):
224 # find the widget for each node in scene
225 sliceNode = slicer.mrmlScene.GetNthNodeByClass(nodeIndex, "vtkMRMLSliceNode")
226 sliceWidget = layoutManager.sliceWidget(sliceNode.GetLayoutName())
227 if sliceWidget:
228 # add observers and keep track of tags
229 style = sliceWidget.sliceView().interactorStyle()
230 self.sliceWidgetsPerStyle[style] = sliceWidget
231 events = ("MouseMoveEvent", "EnterEvent", "LeaveEvent")
232 for event in events:
233 tag = style.AddObserver(event, self.processEvent)
234 self.styleObserverTags.append([style,tag])
235
236 def removeObservers(self):
237 for observee,tag in self.styleObserverTags:
238 observee.RemoveObserver(tag)
239 self.styleObserverTags = []
240 self.sliceWidgetsPerStyle = {}
