214template <typename T>
215std::vector<T*> vtkSlicerLayerDMLogic::GetReferenceNodes(vtkMRMLNode* node, const char* role)
216{
217 if (!node)
218 {
219 return {};
220 }
221
222 std::vector<T*> referenceNodes;
223 for (int i_node = 0; i_node < node->GetNumberOfNodeReferences(role); i_node++)
224 {
225 if (auto refNode = T::SafeDownCast(node->GetNthNodeReference(role, i_node)))
226 {
227 referenceNodes.emplace_back(refNode);
228 }
229 }
230 return referenceNodes;
231}

250template <typename T>
251T* vtkSlicerLayerDMLogic::CreateReferenceNode(vtkMRMLNode* node, const char* role, bool allowMultiple)
252{
253 auto scene = node ? node->GetScene() : nullptr;
254 if (!node || !scene)
255 {
256 return {};
257 }
258
259 auto nodeRef = GetReferenceNode<T>(node, role);
260 if (nodeRef && !allowMultiple)
261 {
262 return nodeRef;
263 }
264
265 nodeRef = T::SafeDownCast(scene->AddNewNodeByClass(vtkSmartPointer<T>::New()->GetClassName()));
266 return AddReferenceNode<T>(node, nodeRef, role);
267}
